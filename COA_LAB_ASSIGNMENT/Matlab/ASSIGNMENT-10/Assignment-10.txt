✅ 13. Matrix Properties (True/False)
A = rand(3);
B = rand(3);
C = rand(3);

% i
isequal(A+B, B+A)        % TRUE
% ii
isequal(A-B, B-A)        % FALSE
% iii
isequal(A+(B+C), (A+B)+C) % TRUE
% iv
isequal(A*(B+C), A*B + A*C) % TRUE
% v
isequal((B+C)/A, B/A + C/A) % TRUE (if A invertible)
% vi
isequal(A*B, B*A)        % FALSE (generally)
% vii
isequal(A*1, A)          % TRUE
% viii
isequal(A*inv(A), eye(3)) % TRUE (if invertible)
% ix
isequal(A, A')           % FALSE (generally)
% x
isequal(A, -A')          % FALSE
